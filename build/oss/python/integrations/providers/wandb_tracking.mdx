```python
"""主函数。

此函数用于尝试回调处理程序。
场景：
1. OpenAI LLM
2. 具有多个子链 (SubChains) 且有多次生成的链 (Chain)
3. 带有工具 (Tools) 的智能体 (Agent)
"""
session_group = datetime.now().strftime("%m.%d.%Y_%H.%M.%S")
wandb_callback = WandbCallbackHandler(
    job_type="inference",
    project="langchain_callback_demo",
    group=f"minimal_{session_group}",
    name="llm",
    tags=["test"],
)
callbacks = [StdOutCallbackHandler(), wandb_callback]
llm = OpenAI(temperature=0, callbacks=callbacks)
```

```
# WandbCallbackHandler.flush_tracker(...) 的默认值

reset: bool = True,

```

`flush_tracker` 函数用于将 LangChain 会话日志记录到 Weights & Biases。它接受 LangChain 模块或智能体，并至少记录提示和生成的内容，以及 LangChain 模块的序列化形式到指定的 Weights & Biases 项目中。默认情况下，我们重置会话而不是直接结束会话。

### 在链 (Chains) 中使用

```python
from langchain.chains import LLMChain
from langchain_core.prompts import PromptTemplate
```

```python
# 场景 2 - 链
template = """你是一位剧作家。给定一部戏剧的标题，你的工作是为该标题撰写剧情简介。
标题: {title}
剧作家: 这是上述剧作的简介:"""
prompt_template = PromptTemplate(input_variables=["title"], template=template)
synopsis_chain = LLMChain(llm=llm, prompt=prompt_template, callbacks=callbacks)

test_prompts = [
    {
        "title": "关于优秀的视频游戏如何推动游戏设计的界限的纪录片"
    },
    {"title": "可卡因熊大战海洛因狼"},
    {"title": "一流的 MLOps 工具"},
]
synopsis_chain.apply(test_prompts)
wandb_callback.flush_tracker(synopsis_chain, name="agent")
```

### 配合智能体 (Agents) 使用

```python
from langchain.agents import AgentType, initialize_agent, load_tools
```

```python
# 场景 3 - 带有工具的智能体
tools = load_tools(["serpapi", "llm-math"], llm=llm)
agent = initialize_agent(
    tools,
    llm,
    agent=AgentType.ZERO_SHOT_REACT_DESCRIPTION,
)
agent.run(
    "莱昂纳多·迪卡普里奥的女友是谁？她现在的年龄的 0.43 次方是多少？",
    callbacks=callbacks,
)
wandb_callback.flush_tracker(agent, reset=False, finish=True)
```
