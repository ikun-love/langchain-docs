title: Perigon（皮里贡）
---

Perigon API 套件提供快速、结构化的方式来访问全球新闻和事件数据，帮助您构建实时、数据驱动的产品。无论您是在跟踪新兴风险、挖掘相关文章，还是揭示关键见解，Perigon 都能为您提供以编程方式完成这些任务的工具。

与传统的基于关键字的搜索不同，Perigon 的**语义搜索**功能使其能够理解查询的上下文并返回最相关的文档。

本指南（Notebook）演示了如何将 Perigon 的检索器与 LangChain 结合使用，分别用于新闻文章和维基百科内容。

## 设置 (Setup)

### 安装 (Installation)

安装 LangChain Perigon 集成包：

```python
pip install -qU langchain-perigon

# 本指南所需的其他依赖项
pip install -qU langchain langchain-openai langchain-community
```

### 凭证 (Credentials)

要使用此集成，您需要一个 Perigon API 密钥。请访问 [Perigon.io](https://perigon.io/) 注册并获取您的 API 密钥。

```python
import getpass
import os

if not os.environ.get("PERIGON_API_KEY"):
    os.environ["PERIGON_API_KEY"] = getpass.getpass("Perigon API 密钥:\n")
```

## 使用 ArticlesRetriever (文章检索器)

`ArticlesRetriever` 允许您利用语义搜索能力来搜索新闻文章：

### 基本用法 (Basic Usage)

```python
from langchain_perigon import ArticlesRetriever

# 创建 ArticlesRetriever 的新实例
# PERIGON_API_KEY 会自动从环境变量中读取
retriever = ArticlesRetriever()

try:
    # 使用语义搜索查询文章
    documents = retriever.invoke("artificial intelligence developments")

    # 检查是否获得了结果
    if not documents:
        print("未找到与给定查询相关的文章。")
    else:
        print(f"找到了 {len(documents)} 篇文章")

        # 显示前 3 个结果及元数据
        for doc in documents[:3]:
            # 安全地提取元数据并提供回退值
            print(f"标题 (Title): {doc.metadata.get('title', 'N/A')}")
            print(f"URL: {doc.metadata.get('url', 'N/A')}")
            print(f"发布时间 (Published): {doc.metadata.get('publishedAt', 'N/A')}")
            print(f"内容 (Content): {doc.page_content[:200]}...")
            print("-" * 80)
except Exception as e:
    print(f"检索文章时发生错误: {e}")
```

### 带有过滤的高级功能 (Advanced Features with Filtering)

您可以使用高级过滤选项来缩小搜索结果范围：

```python
from langchain_perigon import ArticlesRetriever, ArticlesFilter

# 使用自定义参数创建检索器
# PERIGON_API_KEY 会自动从环境变量中读取
retriever = ArticlesRetriever(
    k=10  # 返回结果的数量
)

# 定义高级过滤选项
options: ArticlesFilter = {
    "size": 10,
    "showReprints": False,  # 排除转载文章
    "filter": {
        "country": "us",  # 仅限美国文章
        "category": "tech",  # 技术类别
        "source": ["techcrunch.com", "wired.com"]  # 指定来源
    }
}

try:
    # 应用高级过滤器进行搜索
    documents = retriever.invoke("machine learning breakthroughs", options=options)

    if not documents:
        print("未找到符合过滤条件的文章。")
    else:
        print(f"找到了 {len(documents)} 篇经过筛选的文章")

        # 显示带有相关元数据的结果
        for doc in documents[:3]:
            print(f"标题 (Title): {doc.metadata.get('title', 'N/A')}")
            print(f"来源 (Source): {doc.metadata.get('source', 'N/A')}")
            print(f"类别 (Category): {doc.metadata.get('category', 'N/A')}")
            print(f"内容 (Content): {doc.page_content[:150]}...")
            print("-" * 80)

except Exception as e:
    print(f"检索过滤后的文章时发生错误: {e}")
```

### 基于位置的过滤 (Location-Based Filtering)

您可以按地理位置相关性过滤文章：

```python
from langchain_perigon.types import ArticlesFilter
from langchain_perigon import ArticlesRetriever

retriever = ArticlesRetriever()

# 按地点过滤
location_options: ArticlesFilter = {
    "size": 5,
    "filter": {"country": "us", "state": "CA", "city": "San Francisco"},
}

documents = retriever.invoke("startup funding rounds", options=location_options)

print(f"找到了 {len(documents)} 篇关于旧金山初创公司融资的文章")
for doc in documents:
    print(f"标题 (Title): {doc.metadata.get('title', 'N/A')}")
    print("-" * 60)
```

## 使用 WikipediaRetriever (维基百科检索器)

`WikipediaRetriever` 提供了针对维基百科内容的语义搜索能力，并提供丰富的元数据：

### 基本用法 (Basic Usage)

```python
from langchain_perigon import WikipediaRetriever

# 创建 WikipediaRetriever 的新实例
# PERIGON_API_KEY 会自动从环境变量中读取
wiki_retriever = WikipediaRetriever()

try:
    # 使用语义搜索查询维基百科文章
    documents = wiki_retriever.invoke("quantum computing")

    # 处理前验证结果
    if not documents:
        print("未找到与给定查询相关的维基百科文章。")
    else:
        print(f"找到了 {len(documents)} 篇维基百科文章")

        # 显示前 3 个结果及丰富的元数据
        for doc in documents[:3]:
            # 提取特定于维基百科的元数据
            print(f"标题 (Title): {doc.metadata.get('title', 'N/A')}")
            print(f"页面浏览量 (Pageviews): {doc.metadata.get('pageviews', 'N/A')}")
            print(f"Wikidata ID: {doc.metadata.get('wikidataId', 'N/A')}")
            print(f"内容 (Content): {doc.page_content[:200]}...")
            print("-" * 80)
except Exception as e:
    print(f"检索维基百科文章时发生错误: {e}")
```

### 高级维基百科搜索 (Advanced Wikipedia Search)

您可以按流行度、类别和其他元数据过滤维基百科结果：

```python
from langchain_perigon import WikipediaRetriever, WikipediaOptions

# 使用自定义参数创建检索器
# PERIGON_API_KEY 会自动从环境变量中读取
wiki_retriever = WikipediaRetriever(k=5)

# 定义高级过滤选项
wiki_options: WikipediaOptions = {
    "size": 5,
    "pageviewsFrom": 100,  # 仅限日浏览量达到 100 次以上的热门页面
    "filter": {
        "wikidataInstanceOfLabel": ["academic discipline"],  # 学科实例
        "category": ["Computer science", "Physics"],  # 计算机科学或物理类别
    },
}

# 应用过滤器进行搜索
documents = wiki_retriever.invoke("machine learning", options=wiki_options)

print(f"找到了 {len(documents)} 篇学术性的维基百科文章")
for doc in documents:
    print(f"标题 (Title): {doc.metadata.get('title', 'N/A')}")
    print(f"每日浏览量 (Daily pageviews): {doc.metadata.get('pageviews', 'N/A')}")
    print(f"实例类型 (Instance of): {doc.metadata.get('wikidataInstanceOf', 'N/A')}")
    print(f"Wiki 代码 (Wiki code): {doc.metadata.get('wikiCode', 'N/A')}")
    print("-" * 80)
```

### 基于时间的维基百科过滤 (Time-Based Wikipedia Filtering)

按修订日期过滤维基百科文章：

```python
from langchain_perigon import WikipediaRetriever, WikipediaOptions

wiki_retriever = WikipediaRetriever()

# 按近期修订版本过滤
recent_options: WikipediaOptions = {
    "size": 10,
    "wiki_revision_from": "2025-09-22T00:00:00.000",  # 最近更新的文章
    "filter": {"with_pageviews": True},  # 仅包含有页面浏览量数据的文章
}

documents = wiki_retriever.invoke("artificial intelligence", options=recent_options)

print(f"找到了 {len(documents)} 篇最近更新的 AI 文章")
for doc in documents:
    print(f"标题 (Title): {doc.metadata.get('title', 'N/A')}")
    print(f"最后修订时间 (Last revision): {doc.metadata.get('wikiRevisionTs', 'N/A')}")
    print(f"浏览量 (Pageviews): {doc.metadata.get('pageviews', 'N/A')}")
    print("-" * 60)
```

## 异步用法 (Async Usage)

两种检索器都支持异步操作，以提高性能：

```python
import asyncio
from langchain_perigon import (
    ArticlesRetriever,
    WikipediaRetriever,
    ArticlesFilter,
    WikipediaOptions,
)


async def search_both():
    """执行跨新闻文章和维基百科的并发搜索。

    返回:
        tuple: (news_articles, wikipedia_docs) - 来自两个检索器的结果

    引发:
        Exception: 如果任一检索器失败或发生 API 错误
    """
    # 初始化检索器，自动检测 API 密钥
    articles_retriever = ArticlesRetriever()
    wiki_retriever = WikipediaRetriever()

    # 配置搜索选项以获取目标结果
    articles_options: ArticlesFilter = {
        "size": 3,  # 限制为 3 篇文章以加快响应速度
        "filter": {
            "country": "us",      # 美国新闻来源
            "category": "tech",   # 仅限技术类别
        },
    }

    # 按流行度（浏览量）过滤维基百科结果
    wiki_options: WikipediaOptions = {
        "size": 3,           # 限制为 3 篇文章
        "pageviewsFrom": 50  # 仅限每日浏览量达到 50 次以上的文章
    }

    try:
        # 执行并发异步搜索以提高性能
        articles_task = articles_retriever.ainvoke(
            "climate change", options=articles_options
        )
        wiki_task = wiki_retriever.ainvoke(
            "climate change", options=wiki_options
        )

        # 等待两个搜索同时完成
        articles, wiki_docs = await asyncio.gather(
            articles_task, wiki_task, return_exceptions=True
        )

        # 处理任一检索器可能出现的异常
        if isinstance(articles, Exception):
            print(f"文章检索失败: {articles}")
            articles = []
        if isinstance(wiki_docs, Exception):
            print(f"维基百科检索失败: {wiki_docs}")
            wiki_docs = []

        return articles, wiki_docs

    except Exception as e:
        print(f"并发搜索中发生错误: {e}")
        return [], []


# 运行异步搜索并进行错误处理
try:
    articles, wiki_docs = asyncio.run(search_both())

    # 显示结果摘要
    print(f"找到了 {len(articles)} 篇新闻文章和 {len(wiki_docs)} 篇维基百科文章")

    # 如果有结果，则显示样本结果
    if articles:
        print(f"样本文章标题: {articles[0].metadata.get('title', 'N/A')}")
    if wiki_docs:
        print(f"样本维基百科标题: {wiki_docs[0].metadata.get('title', 'N/A')}")

except Exception as e:
    print(f"异步搜索失败: {e}")
```
